cmake_minimum_required(VERSION 3.19)
project(RecvtConsole LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
# Local Qt 6 desktop kit. Explicit Qt6_DIR/CMAKE_PREFIX_PATH always takes precedence.
if(WIN32 AND NOT DEFINED Qt6_DIR AND NOT CMAKE_PREFIX_PATH)
    if(MSVC)
        set(_recvt_local_qt "D:/qt6/6.11.1/msvc2022_64")
    elseif(MINGW)
        set(_recvt_local_qt "D:/qt6/6.11.1/mingw_64")
    endif()
    if(EXISTS "${_recvt_local_qt}/lib/cmake/Qt6/Qt6Config.cmake")
        list(PREPEND CMAKE_PREFIX_PATH "${_recvt_local_qt}")
    endif()
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network 3DCore 3DRender 3DExtras)
if(MINGW AND Qt6_DIR MATCHES "msvc")
    message(FATAL_ERROR "MinGW requires a MinGW Qt kit. Set CMAKE_PREFIX_PATH=D:/qt6/6.11.1/mingw_64 and reset the CMake cache.")
endif()
option(RECVT_BUILD_TESTS "Build local protocol tests" OFF)
if(RECVT_BUILD_TESTS)
    enable_testing()
    add_executable(recvt_protocol_test protocol_test.cpp script_executor.cpp script_executor.h llm_client.cpp)
    target_include_directories(recvt_protocol_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(recvt_protocol_test PRIVATE Qt6::Core)
    if(MSVC)
        target_compile_options(recvt_protocol_test PRIVATE /utf-8)
    endif()
    add_test(NAME recvt_protocol COMMAND recvt_protocol_test)
    add_executable(recvt_preview_test preview_test.cpp preview_controller.cpp)
    target_include_directories(recvt_preview_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(recvt_preview_test PRIVATE Qt6::Widgets Qt6::3DRender)
    add_test(NAME recvt_preview COMMAND recvt_preview_test)
    set_tests_properties(recvt_preview PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(WIN32)
        # The runtime deployment only installs qwindows; the headless test needs
        # the SDK's offscreen plugin from the same Qt kit.
        get_filename_component(_recvt_qt_plugins "${Qt6_DIR}/../../../plugins" ABSOLUTE)
        set_tests_properties(recvt_preview PROPERTIES ENVIRONMENT
            "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${_recvt_qt_plugins}")
    endif()
    add_executable(recvt_error_guard_test error_guard_test.cpp error_guard.cpp)
    target_include_directories(recvt_error_guard_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(recvt_error_guard_test PRIVATE Qt6::Core)
    if(MSVC)
        target_compile_options(recvt_error_guard_test PRIVATE /utf-8)
    endif()
    add_test(NAME recvt_error_guard COMMAND recvt_error_guard_test)
    add_executable(recvt_ssl_probe ssl_probe.cpp)
    target_link_libraries(recvt_ssl_probe PRIVATE Qt6::Network)
endif()
add_executable(RecvtConsole WIN32 MACOSX_BUNDLE
    main.cpp mainwindow.cpp mainwindow.h types.h error_guard.cpp error_guard.h
    llm_client.h llm_client.cpp ollama_llm.h ollama_llm.cpp
    deepseek_llm.h deepseek_llm.cpp script_executor.h script_executor.cpp preview_controller.h preview_controller.cpp)
if(WIN32 AND MINGW)
    # Some Windows MinGW moc builds cannot create output under non-ASCII directories.
    string(MD5 _recvt_build_id "${CMAKE_BINARY_DIR}")
    file(TO_CMAKE_PATH "$ENV{TEMP}" _recvt_temp)
    set_target_properties(RecvtConsole PROPERTIES
        AUTOGEN_BUILD_DIR "${_recvt_temp}/RecvtConsole-autogen-${_recvt_build_id}")
endif()
foreach(component Core Widgets Network 3DCore 3DRender 3DExtras)
    target_link_libraries(RecvtConsole PRIVATE Qt6::${component})
endforeach()
if(MSVC)
    target_compile_options(RecvtConsole PRIVATE /utf-8)
endif()
option(RECVT_DEPLOY_RUNTIME "Deploy Qt runtime beside the Windows executable" ON)
if(WIN32 AND RECVT_DEPLOY_RUNTIME)
    find_program(RECVT_WINDEPLOYQT windeployqt HINTS "${Qt6_DIR}/../../../bin" NO_DEFAULT_PATH)
    if(RECVT_WINDEPLOYQT)
        if(MINGW)
            # This MinGW kit supplies release Qt DLLs for both application configurations.
            set(_recvt_deploy_mode --release)
        else()
            set(_recvt_deploy_mode "$<IF:$<CONFIG:Debug>,--debug,--release>")
        endif()
        add_custom_command(TARGET RecvtConsole POST_BUILD
            COMMAND "${RECVT_WINDEPLOYQT}" --no-translations --no-compiler-runtime
                "${_recvt_deploy_mode}" "$<TARGET_FILE:RecvtConsole>"
            COMMENT "Deploying Qt 6 runtime" VERBATIM)
        if(MINGW)
            get_filename_component(_recvt_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
            foreach(_recvt_runtime libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
                if(EXISTS "${_recvt_compiler_bin}/${_recvt_runtime}")
                    add_custom_command(TARGET RecvtConsole POST_BUILD
                        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                            "${_recvt_compiler_bin}/${_recvt_runtime}" "$<TARGET_FILE_DIR:RecvtConsole>"
                        VERBATIM)
                endif()
            endforeach()
        endif()
    endif()
endif()
install(TARGETS RecvtConsole BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
